from decimal import Decimal, InvalidOperation, ROUND_HALF_UP

def quantidade(texto):
    """Converte e valida o texto da quantidade para um número inteiro positivo."""
    try:
        qtd = int(str(texto).strip())
        if qtd <= 0:
            raise ValueError("A quantidade deve ser maior que zero.")
        return qtd
    except (ValueError, TypeError) as e:
        raise ValueError("Quantidade inválida.") from e


def formatar(centavos):
    """Formata um valor inteiro em centavos para a representação em moeda (ex: R$ 10,50)."""
    try:
        valor = Decimal(int(centavos)) / Decimal(100)
        valor = valor.quantize(Decimal('0.01'), rounding=ROUND_HALF_UP)
        # Formato brasileiro: milhar com ponto e decimal com vírgula (ex: R$ 1.234,50)
        texto = f"{valor:,.2f}".replace(",", "X").replace(".", ",").replace("X", ".")
        return f"R$ {texto}"
    except (ValueError, TypeError, InvalidOperation):
        return "R$ 0,00"


class Produto:
    def __init__(self, codigo, nome, preco_centavos):
        self.codigo = int(codigo)
        self.nome = str(nome).strip()
        self.preco_centavos = int(preco_centavos)  # Garante que guarda o valor inteiro em centavos
        if self.preco_centavos < 0:
            raise ValueError("O preço em centavos não pode ser negativo.")


class Item:
    def __init__(self, produto, texto_quantidade):
        self.produto = produto
        self.quantidade = quantidade(texto_quantidade)

    def subtotal(self):
        return self.produto.preco_centavos * self.quantidade


class Pedido:
    def __init__(self):
        self._itens = {}  # Mapeia codigo_produto -> Item
        self._aberto = True
        self._forma_pagamento = None
        self._confirmado = False

    def exigir_aberto(self):
        if not self._aberto:
            raise RuntimeError("O pedido já foi finalizado.")

    def adicionar(self, produto, texto_quantidade):
        self.exigir_aberto()
        qtd = quantidade(texto_quantidade)
        codigo = produto.codigo
        if codigo in self._itens:
            self._itens[codigo].quantidade += qtd
        else:
            self._itens[codigo] = Item(produto, qtd)

    def total_centavos(self):
        return sum(item.subtotal() for item in self._itens.values())

    def linhas(self):
        resultado = []
        for item in self._itens.values():
            resultado.append({
                "codigo": item.produto.codigo,
                "nome": item.produto.nome,
                "preco_unitario": item.produto.preco_centavos,
                "quantidade": item.quantidade,
                "subtotal": item.subtotal()
            })
        return resultado

    def atualizar(self, codigo, texto_quantidade):
        self.exigir_aberto()
        cod = int(codigo)
        if cod not in self._itens:
            raise KeyError("Item não encontrado no pedido.")
        qtd = quantidade(texto_quantidade)
        self._itens[cod].quantidade = qtd

    def remover(self, codigo):
        self.exigir_aberto()
        cod = int(codigo)
        if cod in self._itens:
            del self._itens[cod]

    def finalizar(self, forma, confirmado):
        self.exigir_aberto()
        if not self._itens:
            raise ValueError("Não é possível finalizar um pedido vazio.")
        self._forma_pagamento = str(forma).strip()
        self._confirmado = bool(confirmado)
        if self._confirmado:
            self._aberto = False

    def novo(self):
        self._itens.clear()
        self._aberto = True
        self._forma_pagamento = None
        self._confirmado = False

from decimal import Decimal, InvalidOperation, ROUND_HALF_UP
import re

FATORES = {'Dinheiro': '0.87', 'PIX': '0.87', 'Débito': '0.92', 'Crédito 1x': '1.00'}

def valor_monetario(texto, permitir_zero=True):
    """Valida e converte uma string de texto em um objeto Decimal monetário positivo."""
    if texto is None:
        raise ValueError("Texto vazio ou inválido.")
    
    # Remove espaços e formatações comuns de moeda (ex: 'R$', pontos de milhar)
    limpo = str(texto).strip()
    if limpo.upper().startswith('R$'):
        limpo = limpo[2:].strip()
    
    # Substitui ponto por vazio (milhar) e vírgula por ponto (decimal) se necessário
    if ',' in limpo and '.' in limpo:
        limpo = limpo.replace('.', '').replace(',', '.')
    elif ',' in limpo:
        limpo = limpo.replace(',', '.')

    try:
        valor = Decimal(limpo)
        if valor < 0:
            raise ValueError("O valor monetário não pode ser negativo.")
        if not permitir_zero and valor == 0:
            raise ValueError("O valor monetário deve ser maior que zero.")
        return valor.quantize(Decimal('0.01'), rounding=ROUND_HALF_UP)
    except (InvalidOperation, ValueError) as e:
        raise ValueError("Formato de valor monetário inválido.") from e


def preco_para_centavos(texto):
    """Converte um texto de preço ou valor monetário para um inteiro em centavos."""
    valor = valor_monetario(texto, permitir_zero=True)
    return int((valor * 100).quantize(Decimal('1'), rounding=ROUND_HALF_UP))


def calcular_pagamento(subtotal_texto, forma):
    """Calcula o valor final aplicando o fator de desconto/acréscimo correspondente à forma de pagamento."""
    if forma not in FATORES:
        raise ValueError(f"Forma de pagamento inválida: {forma}")
    
    subtotal = valor_monetario(subtotal_texto, permitir_zero=False)
    fator = Decimal(FATORES[forma])
    
    total_calculado = subtotal * fator
    total_centavos = int((total_calculado * 100).quantize(Decimal('1'), rounding=ROUND_HALF_UP))
    
    # Retorna formatado ou em centavos/decimal dependendo do contrato anterior; 
    # mantendo compatibilidade com o retorno em Decimal quantizado:
    return (Decimal(total_centavos) / Decimal(100)).quantize(Decimal('0.01'), rounding=ROUND_HALF_UP)

from pathlib import Path
import json
import os
import tempfile

def validar_catalogo(dados):
    """Valida a estrutura completa de um catálogo de produtos carregado via JSON.
       Deve ser uma lista contendo dicionários com chaves: 'id', 'nome', 'preco_centavos'.
       Garante que IDs sejam inteiros positivos únicos, nomes sejam textos válidos 
       e preços sejam inteiros positivos. Retorna True se válido, ou levanta ValueError.
    """
    if not isinstance(dados, list):
        raise ValueError("O catálogo deve ser uma lista de produtos.")
    
    ids_vistos = set()
    for item in dados:
        if not isinstance(item, dict):
            raise ValueError("Cada produto no catálogo deve ser um objeto JSON (dicionário).")
        
        # Verifica chaves obrigatórias
        for chave in ("id", "nome", "preco_centavos"):
            if chave not in item:
                raise ValueError(f"Chave obrigatória ausente no produto: '{chave}'.")
        
        # Validação do ID
        try:
            codigo = int(item["id"])
        except (ValueError, TypeError) as e:
            raise ValueError("O ID do produto deve ser um número inteiro.") from e
            
        if codigo <= 0:
            raise ValueError("O ID do produto deve ser maior que zero.")
            
        if codigo in ids_vistos:
            raise ValueError(f"ID de produto duplicado encontrado: {codigo}.")
        ids_vistos.add(codigo)
        
        # Validação do Nome
        nome = str(item["nome"]).strip()
        if not nome:
            raise ValueError("O nome do produto não pode estar vazio.")
            
        # Validação do Preço em Centavos
        try:
            preco = int(item["preco_centavos"])
        except (ValueError, TypeError) as e:
            raise ValueError("O preço em centavos deve ser um número inteiro.") from e
            
        if preco < 0:
            raise ValueError("O preço em centavos não pode ser negativo.")
            
    return True


def ler_catalogo(caminho):
    """Lê um catálogo de produtos a partir de um arquivo JSON.
       - Se o arquivo não existir, retorna uma lista vazia sem criar arquivo.
       - Se o arquivo contiver JSON inválido ou estrutura inválida, levanta exceção 
         preservando integralmente os bytes originais do arquivo.
    """
    path = Path(caminho)
    if not path.exists():
        return []
    
    # Lê o texto com codificação UTF-8 explícita
    texto = path.read_text(encoding="utf-8")
    
    try:
        dados = json.loads(texto)
    except json.JSONDecodeError as e:
        raise ValueError(f"Erro de sintaxe no JSON: {e}") from e
        
    # Valida a estrutura de acordo com as regras de negócio do catálogo
    validar_catalogo(dados)
    return dados


def salvar_catalogo(caminho, dados):
    """Salva o catálogo de produtos em um arquivo JSON de forma atômica.
       Valida os dados antes de gravar, escreve em um arquivo temporário no mesmo 
       diretório e só substitui o arquivo de destino ao concluir com sucesso.
    """
    # Valida os dados antes de qualquer escrita
    validar_catalogo(dados)
    
    path = Path(caminho)
    path.parent.mkdir(parents=True, exist_ok=True)
    
    # Serializa os dados em JSON formatado com UTF-8
    texto = json.dumps(dados, ensure_ascii=False, indent=4)
    
    # Escreve em arquivo temporário no mesmo diretório para garantir atomicidade
    dir_destino = path.parent
    fd, temp_path = tempfile.mkstemp(dir=dir_destino, text=True)
    try:
        with os.fdopen(fd, 'w', encoding='utf-8') as f:
            f.write(texto)
        # Substitui o arquivo de destino de forma segura e atômica
        os.replace(temp_path, path)
    except Exception:
        # Garante a remoção do temporário em caso de falha de I/O
        if os.path.exists(temp_path):
            os.remove(temp_path)
        raise

    def acao_adicionar_item(self):
        try:
            txt_id = self.txt_prod_id.get().strip()
            nome = self.txt_prod_nome.get().strip()
            txt_preco = self.txt_prod_preco.get().strip()
            txt_qtd = self.txt_prod_qtd.get().strip()
 
            if not txt_id or not nome or not txt_preco or not txt_qtd:
                raise ValueError("Preencha todos os campos do produto de teste.")
 
            codigo = int(txt_id)
            
        
            preco_centavos = pagamento.preco_para_centavos(txt_preco)
 
            # Instancia o produto passando o valor em centavos correto
            produto = modelos.Produto(codigo, nome, preco_centavos)
            
            # Adiciona ao pedido
            self.pedido.adicionar(produto, txt_qtd)
            
            # Atualiza a tabela na tela imediatamente
            self.atualizar_tabela_carrinho()
            
        except Exception as e:
            messagebox.showerror("Erro ao adicionar", f"Não foi possível lançar o item:\n{e}")
 
    def atualizar_tabela_carrinho(self):
        # Limpa a tabela atual
        for item in self.tree_carrinho.get_children():
            self.tree_carrinho.delete(item)
 
        # Insere cada linha obtida do modelo de dados
        for linha in self.pedido.linhas():
            # Convertemos explicitamente os valores para garantir que a formatação receba o inteiro correto
            preco_formatado = modelos.formatar(linha["preco_unitario"])
            subtotal_formatado = modelos.formatar(linha["subtotal"])
 
            self.tree_carrinho.insert("", "end", values=(
                linha["codigo"],
                linha["nome"],
                preco_formatado,
                linha["quantidade"],
                subtotal_formatado
            ))
 
        total_cent = self.pedido.total_centavos()
        self.lbl_total.configure(text=f"Total do Pedido: {modelos.formatar(total_cent)}")
 
    def acao_novo_pedido(self):
        if messagebox.askyesno("Confirmar", "Deseja limpar e iniciar um novo pedido?"):
            self.pedido.novo()
            self.atualizar_tabela_carrinho()
            messagebox.showinfo("Novo Pedido", "Carrinho reiniciado com sucesso.")

    def acao_carregar_catalogo(self):
        caminho = filedialog.askopenfilename(
            title="Selecione o catálogo JSON",
            filetypes=[("Arquivos JSON", "*.json"), ("Todos os arquivos", "*.*")]
        )
        if not caminho:
            self.lbl_status_json.configure(text="Ação de carregar cancelada pelo usuário.", foreground="blue")
            return
 
        try:
            dados = arquivos.ler_catalogo(caminho)
            self.catalogo_atual = dados
            self.caminho_arquivo_atual = caminho
 
            import json
            texto_formatado = json.dumps(dados, ensure_ascii=False, indent=4)
            self.txt_json_conteudo.delete("1.0", tk.END)
            self.txt_json_conteudo.insert("1.0", texto_formatado)
 
            self.lbl_status_json.configure(text=f"Sucesso: Catálogo carregado de '{Path(caminho).name}'.", foreground="green")
            messagebox.showinfo("Sucesso", f"Catálogo carregado com {len(dados)} produtos!")
            
        except ValueError as e:
            self.lbl_status_json.configure(text=f"Erro de validação/leitura: {e}", foreground="red")
            messagebox.showerror("Erro no Catálogo", f"Não foi possível carregar o arquivo:\n{e}\n\nO arquivo original foi preservado.")
        except Exception as e:
            self.lbl_status_json.configure(text=f"Erro inesperado: {e}", foreground="red")
            messagebox.showerror("Erro", str(e))
 
    def acao_salvar_catalogo(self):
        caminho = filedialog.asksaveasfilename(
            title="Salvar catálogo JSON",
            defaultextension=".json",
            filetypes=[("Arquivos JSON", "*.json")]
        )
        if not caminho:
            return
 
        dados_exemplo = [
            {"id": 1, "nome": "Picolé de Limão", "preco_centavos": 450},
            {"id": 2, "nome": "Açaí Tradicional", "preco_centavos": 1200},
            {"id": 3, "nome": "Água Mineral 500ml", "preco_centavos": 300}
        ]
 
        try:
            arquivos.salvar_catalogo(caminho, dados_exemplo)
            self.lbl_status_json.configure(text=f"Catálogo salvo com sucesso em '{Path(caminho).name}'.", foreground="green")
            messagebox.showinfo("Sucesso", "Arquivo de catálogo salvo com segurança!")
        except Exception as e:
            self.lbl_status_json.configure(text=f"Erro ao salvar: {e}", foreground="red")
            messagebox.showerror("Erro", str(e))
