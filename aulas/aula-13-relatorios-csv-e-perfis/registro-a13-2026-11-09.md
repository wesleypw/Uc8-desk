def quantidade(texto):
    """Converte e valida o texto de quantidade para inteiro positivo (> 0)."""
    try:
        val = int(str(texto).strip())
        if val <= 0:
            raise ValueError("A quantidade deve ser maior que zero.")
        return val
    except (ValueError, TypeError):
        raise ValueError("Quantidade inválida. Informe um número inteiro positivo.")


def formatar(centavos):
    """Formata valor em centavos (inteiro) para string no padrão monetário brasileiro (ex: R$ 12,50)."""
    if not isinstance(centavos, int):
        try:
            centavos = int(centavos)
        except (ValueError, TypeError):
            raise ValueError("Valor em centavos deve ser numérico.")
    reais = centavos / 100
    return f"R$ {reais:,.2f}".replace(",", "X").replace(".", ",").replace("X", ".")


class Produto:
    def __init__(self, codigo, nome, preco_centavos):
        if not str(codigo).strip():
            raise ValueError("Código do produto é obrigatório.")
        if not str(nome).strip():
            raise ValueError("Nome do produto é obrigatório.")
        if not isinstance(preco_centavos, int) or preco_centavos < 0:
            raise ValueError("Preço em centavos deve ser um inteiro não negativo.")
            
        self.codigo = str(codigo).strip()
        self.nome = str(nome).strip()
        self.preco_centavos = preco_centavos


class Pizza(Produto):
    def __init__(self, codigo, nome, tamanho, preco_centavos):
        super().__init__(codigo, nome, preco_centavos)
        if not str(tamanho).strip():
            raise ValueError("Tamanho é obrigatório.")
        self.tamanho = str(tamanho).strip()


class Bebida(Produto):
    def __init__(self, codigo, nome, preco_centavos):
        super().__init__(codigo, nome, preco_centavos)


class Item:
    def __init__(self, produto, texto_quantidade):
        if not isinstance(produto, Produto):
            raise ValueError("Objeto produto inválido.")
        self.produto = produto
        self.qtd = quantidade(texto_quantidade)

    def subtotal(self):
        """Retorna o subtotal do item em centavos."""
        return self.produto.preco_centavos * self.qtd


class Pedido:
    def __init__(self):
        self.itens = {}  # Mapeia codigo_produto -> Instância de Item
        self.status = "ABERTO"  # Estados: ABERTO, FINALIZADO
        self.comprovante = None

    def exigir_aberto(self):
        """Garante que operações de modificação só ocorram em pedidos abertos."""
        if self.status != "ABERTO":
            raise ValueError("Operação não permitida: o pedido já foi finalizado.")

    def adicionar(self, produto, texto_quantidade):
        self.exigir_aberto()
        qtd_valida = quantidade(texto_quantidade)
        cod = produto.codigo
        
        if cod in self.itens:
            nova_qtd = self.itens[cod].qtd + qtd_valida
            self.itens[cod] = Item(produto, nova_qtd)
        else:
            self.itens[cod] = Item(produto, qtd_valida)

    def total_centavos(self):
        """Calcula a soma de todos os subtotais dos itens do pedido."""
        return sum(item.subtotal() for item in self.itens.values())

    def linhas(self):
        """Retorna a lista com a representação tabular ou estruturada dos itens."""
        resultado = []
        for item in self.itens.values():
            resultado.append({
                "codigo": item.produto.codigo,
                "nome": item.produto.nome,
                "quantidade": item.qtd,
                "preco_unitario": item.produto.preco_centavos,
                "subtotal": item.subtotal()
            })
        return resultado

    def atualizar(self, codigo, texto_quantidade):
        self.exigir_aberto()
        cod = str(codigo).strip()
        if cod not in self.itens:
            raise ValueError(f"Item com código '{cod}' não encontrado no pedido.")
        qtd_valida = quantidade(texto_quantidade)
        produto = self.itens[cod].produto
        self.itens[cod] = Item(produto, qtd_valida)

    def remover(self, codigo):
        self.exigir_aberto()
        cod = str(codigo).strip()
        if cod not in self.itens:
            raise ValueError(f"Item com código '{cod}' não encontrado no pedido.")
        del self.itens[cod]

    def finalizar(self, forma, confirmado=True):
        """
        Finaliza o pedido se houver itens e gera o comprovante.
        Preserva a estrutura do comprovante para consumo por relatorios.py.
        """
        self.exigir_aberto()
        if not self.itens:
            raise ValueError("Não é possível finalizar um pedido vazio.")
        if not confirmado:
            raise ValueError("A confirmação do pagamento é necessária.")

        self.status = "FINALIZADO"
        total = self.total_centavos()
        
        # Estrutura do comprovante confirmada para exportação em relatorios.py
        self.comprovante = {
            "forma_pagamento": forma,
            "total_centavos": total,
            "total_formatado": formatar(total),
            "itens": [
                {
                    "codigo": item.produto.codigo,
                    "nome": item.produto.nome,
                    "quantidade": item.qtd,
                    "preco_unitario_centavos": item.produto.preco_centavos,
                    "subtotal_centavos": item.subtotal()
                }
                for item in self.itens.values()
            ]
        }
        return self.comprovante

    def novo(self):
        """Reinicia a instância do pedido para um novo ciclo de compra."""
        self.itens = {}
        self.status = "ABERTO"
        self.comprovante = None

from decimal import Decimal, InvalidOperation, ROUND_HALF_UP
import re

FATORES = {'Dinheiro': '0.87', 'PIX': '0.87', 'Débito': '0.92', 'Crédito 1x': '1.00'}

def valor_monetario(texto, permitir_zero=True):
    """
    Valida e converte um texto monetário para um objeto Decimal arredondado a 2 casas.
    Aceita formatos como '10,50', 'R$ 10.50', '10.5', etc.
    """
    if texto is None:
        raise ValueError("Texto monetário não pode ser vazio.")
    
    # Remove espaços e símbolos de moeda
    limpo = str(texto).strip()
    limpo = re.sub(r'^[a-zA-Z$R\s]+', '', limpo).strip()
    if not limpo:
        raise ValueError("Texto monetário vazio ou inválido.")

    # Normalização de separadores de milhares/decimais
    # Caso contenha pontos e vírgulas (ex: 1.234,56 ou 1,234.56)
    if ',' in limpo and '.' in limpo:
        if limpo.rfind(',') > limpo.rfind('.'):
            # Formato brasileiro: 1.234,56
            limpo = limpo.replace('.', '').replace(',', '.')
        else:
            # Formato americano: 1,234.56
            limpo = limpo.replace(',', '')
    else:
        # Apenas vírgula como decimal (ex: 12,50)
        limpo = limpo.replace(',', '.')

    try:
        valor = Decimal(limpo)
    except (InvalidOperation, ValueError):
        raise ValueError(f"Formato monetário inválido: '{texto}'.")

    if not permitir_zero and valor <= Decimal('0.00'):
        raise ValueError("O valor deve ser maior que zero.")
    if permitir_zero and valor < Decimal('0.00'):
        raise ValueError("O valor não pode ser negativo.")

    return valor.quantize(Decimal('0.01'), rounding=ROUND_HALF_UP)


def preco_para_centavos(texto):
    """Converte um texto monetário diretamente para um inteiro representando centavos."""
    val_decimal = valor_monetario(texto, permitir_zero=True)
    return int((val_decimal * 100).quantize(Decimal('1'), rounding=ROUND_HALF_UP))


def calcular_pagamento(subtotal_texto, forma):
    """
    Calcula o valor final após aplicar o fator correspondente à forma de pagamento.
    Retorna uma tupla (subtotal_decimal, fator_decimal, total_decimal).
    """
    if forma not in FATORES:
        raise ValueError(f"Forma de pagamento inválida: '{forma}'.")

    subtotal = valor_monetario(subtotal_texto, permitir_zero=False)
    fator = Decimal(FATORES[forma])
    
    total = (subtotal * fator).quantize(Decimal('0.01'), rounding=ROUND_HALF_UP)
    return subtotal, fator, total

def validar_catalogo(dados):
    
    
    if not isinstance(dados, dict):
        raise ValueError("O catálogo raiz deve ser um dicionário.")
    
    # Validações estruturais básicas de campos esperados (pizzas, bebidas, etc.)
    for chave, lista_itens in dados.items():
        if not isinstance(lista_itens, list):
            raise ValueError(f"A categoria '{chave}' deve ser uma lista de itens.")
        for item in lista_itens:
            if not isinstance(item, dict):
                raise ValueError("Cada item do catálogo deve ser um dicionário.")
            if "codigo" not in item or "nome" not in item or "preco_centavos" not in item:
                raise ValueError("Item do catálogo incompleto: faltam campos obrigatórios.")
    return True

def ler_catalogo(caminho):
    
    path_obj = Path(caminho)
    if not path_obj.exists():
        raise FileNotFoundError(f"Arquivo de catálogo não encontrado: '{caminho}'.")
    
    try:
        with open(path_obj, "r", encoding="utf-8") as f:
            dados = json.load(f)
    except json.JSONDecodeError as e:
        raise ValueError(f"Erro ao decodificar JSON do catálogo: {e}")
        
    validar_catalogo(dados)
    return dados


def salvar_catalogo(caminho, dados):
    
    #Salva o catálogo em formato JSON de forma atômica, utilizando um arquivo 
    #temporário no mesmo diretório para evitar corrupção em caso de falhas.
    
    validar_catalogo(dados)
    path_obj = Path(caminho)
    diretorio = path_obj.parent
    diretorio.mkdir(parents=True, exist_ok=True)

    # Escrita atômica via arquivo temporário na mesma pasta
    fd, temp_path = tempfile.mkstemp(dir=str(diretorio), prefix="tmp_cat_", text=True)
    try:
        with os.fdopen(fd, "w", encoding="utf-8") as f:
            json.dump(dados, f, ensure_ascii=False, indent=4)
        
        # Substitui o arquivo original de forma segura
        os.replace(temp_path, path_obj)
    except Exception as e:
        if os.path.exists(temp_path):
            try:
                os.remove(temp_path)
            except OSError:
                pass
        raise RuntimeError(f"Falha ao salvar o catálogo: {e}")

from pathlib import Path
import csv
import os
import tempfile

def exportar_csv(caminho, comprovante):

    if not isinstance(comprovante, dict) or "itens" not in comprovante:
        raise ValueError("Comprovante inválido ou ausente para exportação.")

    path_obj = Path(caminho)
    diretorio = path_obj.parent
    
    # Valida se a pasta de destino existe e é gravável; caso contrário, lança erro antes de corromper
    try:
        diretorio.mkdir(parents=True, exist_ok=True)
    except Exception as e:
        raise PermissionError(f"Não foi possível acessar ou criar o diretório de destino: {e}")

    # Extração de dados do comprovante
    forma_pagamento = comprovante.get("forma_pagamento", "Desconhecida")
    total_centavos = comprovante.get("total_centavos", 0)
    total_formatado = comprovante.get("total_formatado", "R$ 0,00")
    itens = comprovante.get("itens", [])

    # Escreve de forma atômica utilizando um arquivo temporário na mesma pasta
    fd, temp_path = tempfile.mkstemp(dir=str(diretorio), prefix="tmp_rel_", text=True)
    try:
        # newline="" evita linhas em branco extras no Windows; encoding="utf-8-sig" adiciona o BOM para o Excel
        with os.fdopen(fd, "w", encoding="utf-8-sig", newline="") as f:
            writer = csv.writer(f, delimiter=";")
            
            # Cabeçalho do relatório
            writer.writerow(["ID", "Nome", "Quantidade", "Preco_Unitario_Centavos", "Subtotal_Centavos", "Forma_Pagamento", "Total_Pedido_Formatado"])
            
            # Linhas de itens
            for item in itens:
                writer.writerow([
                    item.get("codigo", ""),
                    item.get("nome", ""),
                    item.get("quantidade", 0),
                    item.get("preco_unitario_centavos", 0),
                    item.get("subtotal_centavos", 0),
                    forma_pagamento,
                    total_formatado
                ])
                
        # Substituição atômica do arquivo final
        os.replace(temp_path, path_obj)
    except Exception as e:
        if os.path.exists(temp_path):
            try:
                os.remove(temp_path)
            except OSError:
                pass
        raise RuntimeError(f"Erro ao exportar o relatório CSV: {e}")
    
    return True

class SessaoDemo:
    
    """Simulação local de identificação/perfil; não autentica usuários reais."""

    def __init__(self):
        self.nome = None
        self.perfil = None
        self.ativo = False

    def entrar(self, nome, perfil):
        """
        Define a sessão atual com nome fictício e perfil de demonstração.
        Perfis aceitos tipicamente: 'Caixa', 'Administração'.
        """
        if not str(nome).strip():
            raise ValueError("O nome do usuário não pode ser vazio.")
        if not str(perfil).strip():
            raise ValueError("O perfil de demonstração é obrigatório.")
            
        self.nome = str(nome).strip()
        self.perfil = str(perfil).strip()
        self.ativo = True
        return True

    def pode_configurar(self):
        """
        Verifica se a sessão ativa possui autorização para configurações.
        Apenas o perfil 'Administração' possui permissão nesta simulação.
        """
        if not self.ativo:
            return False
        return self.perfil.lower() in ["administração", "administracao", "admin"]

    def sair(self):
        """Encerra e limpa a sessão fictícia ativa."""
        self.nome = None
        self.perfil = None
        self.ativo = False
        return True

    # Rótulo de título e instruções

    # Frame de Sessão / Perfil
    frame_perfil = ttk.LabelFrame(base, text=" Simulação de Identificação e Perfil ", padding=10)
    frame_perfil.grid(row=1, column=0, sticky="nsew", pady=6)
    frame_perfil.columnconfigure(1, weight=1)

    ttk.Label(frame_perfil, text="Usuário:").grid(row=0, column=0, sticky="w", padx=4)
    entry_nome = ttk.Entry(frame_perfil)
    entry_nome.insert(0, "Operador Caixa")
    entry_nome.grid(row=0, column=1, sticky="ew", padx=4)

    ttk.Label(frame_perfil, text="Perfil:").grid(row=1, column=0, sticky="w", padx=4, pady=4)
    combo_perfil = ttk.Combobox(frame_perfil, values=["Caixa", "Administração"], state="readonly")
    combo_perfil.set("Caixa")
    combo_perfil.grid(row=1, column=1, sticky="ew", padx=4, pady=4)

    lbl_status_sessao = ttk.Label(frame_perfil, text="Sessão: Desconectada", foreground="gray")
    lbl_status_sessao.grid(row=2, column=0, columnspan=2, sticky="w", padx=4, pady=4)

    def acao_entrar():
        nome = entry_nome.get()
        perfil = combo_perfil.get()
        try:
            sessao.entrar(nome, perfil)
            config_status = "Sim (Admin)" if sessao.pode_configurar() else "Não (Caixa)"
            lbl_status_sessao.configure(text=f"Sessão Ativa: {sessao.nome} ({sessao.perfil}) | Pode configurar: {config_status}", foreground="green")
            aviso.configure(text=f"Entrada realizada como {sessao.perfil}.")
        except ValueError as e:
            messagebox.showerror("Erro na Sessão", str(e))

    def acao_sair():
        sessao.sair()
        lbl_status_sessao.configure(text="Sessão: Desconectada", foreground="gray")
        aviso.configure(text="Sessão encerrada com segurança.")

    frame_botoes_sessao = ttk.Frame(frame_perfil)
    frame_botoes_sessao.grid(row=3, column=0, columnspan=2, sticky="ew", pady=(4, 0))
    ttk.Button(frame_botoes_sessao, text="Entrar na Sessão", command=acao_entrar).pack(side="left", padx=2)
    ttk.Button(frame_botoes_sessao, text="Sair", command=acao_sair).pack(side="left", padx=2)

    # Frame de Ações do Pedido e Exportação CSV
    frame_exportacao = ttk.LabelFrame(base, text=" Operações e Exportação CSV ", padding=10)
    frame_exportacao.grid(row=2, column=0, sticky="nsew", pady=6)
    frame_exportacao.columnconfigure(0, weight=1)

    aviso = ttk.Label(frame_exportacao, text="Pedido pronto com item de teste ('Picolé; edição verão').", wraplength=460)
    aviso.grid(row=0, column=0, sticky="w", pady=4)

    def testar_fluxo():
        """Finaliza o pedido, gera o comprovante oficial e exporta para CSV de forma segura."""
        try:
            # Garante que o pedido está finalizado e possui comprovante
            if pedido.status == "ABERTO":
                pedido.finalizar("PIX", confirmado=True)

            # Abre janela de salvamento de arquivo do Tkinter
            arquivo_destino = filedialog.asksaveasfilename(
                defaultextension=".csv",
                filetypes=[("Arquivos CSV", "*.csv"), ("Todos os arquivos", "*.*")],
                title="Exportar Comprovante CSV"
            )

            if not arquivo_destino:
                aviso.configure(text="Exportação cancelada pelo usuário. Pedido preservado.")
                return

            # Executa a exportação atômica
            exportar_csv(arquivo_destino, pedido.comprovante)
            aviso.configure(text=f"Sucesso! Comprovante exportado para: {Path(arquivo_destino).name}")
            messagebox.showinfo("Exportação Concluída", "O relatório CSV foi gerado com sucesso preservando a codificação e as colunas.")

        except Exception as e:
            # Tratamento defensivo de erro sem perda de dados ou estado do pedido
            aviso.configure(text=f"Falha na exportação: {e}")
            messagebox.showerror("Erro ao Exportar", f"Não foi possível salvar o arquivo:\n{e}\n\nO pedido atual permanece intacto.")

    ttk.Button(frame_exportacao, text="Finalizar e Exportar Comprovante CSV", command=testar_fluxo).grid(row=1, column=0, sticky="ew", pady=6)

    # Rodapé informativo
    ttk.Label(base, text="Pressione Tab/Shift+Tab para navegar pelos campos e botões.", font=("Arial", 8), foreground="gray").grid(row=5, column=0, sticky="w", pady=(8, 0))

    return janela


if __name__ == "__main__":
    criar_janela().mainloop()
